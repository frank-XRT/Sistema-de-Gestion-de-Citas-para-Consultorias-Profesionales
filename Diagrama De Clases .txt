%% ===== CONTROL =====

class CControlador {
  +Login()
  +RegistrarCliente()
  +RegistrarConsultor()
  +EliminarConsultor()
  +BuscarConsultoresPorRubro()
  +GenerarHorarios()
  +ReservarCita()
}

class CReporte {
  +CitasPorServicio()
  +ConsultoresPorRubro()
  +IngresosPorTrimestre()
  +ClientesMasFrecuentes()
}

%% ===== PERSONAS =====

class CAdministrador {
  +Codigo: int
  +Nombre: string
  +Dni: string
  +Contraseña: string
}

class CConsultor {
  +Codigo: int
  +Nombre: string
  +Dni: string
  +Sexo: string
  +Telefono: string
  +Correo: string
  +Rubro: string
  +Descripcion: string
  +Contraseña: string
}

class CCliente {
  +Codigo: int
  +Nombre: string
  +Dni: string
  +Sexo: string
  +Telefono: string
  +Correo: string
  +Contraseña: string
}

%% ===== CITAS =====

class CHorarioConsultor {
  +Codigo: int
  +CodigoConsultor: int
  +FechaHoraInicio: DateTime
  +FechaHoraFin: DateTime
  +Estado: string
}

class CCita {
  +Codigo: int
  +CodigoCliente: int
  +CodigoHorario: int
  +Monto: decimal
  +Descripcion: string
  +Estado: string
}

%% ===== RELACIONES PRINCIPALES =====

CConsultor "1" --> "muchos" CHorarioConsultor
CHorarioConsultor "1" --> "0..1" CCita
CCliente "1" --> "muchos" CCita

%% ===== CONTROL =====

CControlador --> CAdministrador
CControlador --> CConsultor
CControlador --> CCliente
CControlador --> CCita

CReporte --> CCita
CReporte --> CConsultor
